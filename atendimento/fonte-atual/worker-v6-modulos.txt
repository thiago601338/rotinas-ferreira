export const RULE_IDS=['direct_answer','no_repeated_questions','singular_plural','short_paragraphs','gentle_correction','moderate_warmth','precise_unavailability','one_useful_question'];
export function editorialRules(profile){
 if(profile?.schemaVersion!==1||profile.enabled!==true||!Array.isArray(profile.ruleIds))return [];

const ACK='(?:ok+|okay|certo|ta certo|esta certo|ta bom|ta bem|tabom|beleza|blz|perfeito|otimo|entendi|entendido|pronto|show|joia|ta joia|uhum|aham)';

const thanksPattern=new RegExp('^(?:'+ACK+' )?'+THANKS+'(?: '+ACK+')?$');
const ackPattern=new RegExp('^'+ACK+'$');
export function courtesyKind(text){
 const n=norm(text);
 if(thanksPattern.test(n))return 'thanks';
 if(ackPattern.test(n)||/^[\s.!👍🙏😊]+$/u.test(String(text??'')))return 'acknowledgement';

}
export function courtesyForTurn(current,history){
 if(!current?.length)return null;
 const latest=Math.max(...current.map(r=>Date.parse(r.created_at)));
 const sent=history.filter(r=>r.direction==='outbound'&&r.raw_payload?.importedHistory!==true&&
  (r.raw_payload?.enviado===true||r.raw_payload?.enviado!==false&&(['conta_da_loja','whatsapp_business_app'].includes(r.raw_payload?.origem)||r.raw_payload?.coexistence===true)))
  .map(r=>Date.parse(r.created_at)).filter(t=>Number.isFinite(t)&&t<=latest);
 const boundary=sent.length?Math.max(...sent):-Infinity;
 const unanswered=current.filter(r=>Date.parse(r.created_at)>boundary);
 if(!unanswered.length||unanswered.some(r=>r.raw_payload?.supported===false||r.raw_payload?.mediaType||r.raw_payload?.storyReference||r.raw_payload?.replyTo))return null;

export function editorialText(text){
 const n=norm(text);
 if(/\b(?:k{3,}|rsrs\w*|ja (?:te |lhe )?(?:falei|disse)|vem logo|o que (?:voce )?esta esperando)\b/.test(n))throw Error('unsafe_editorial_copy');
 let t=String(text??'').replace(/\b(?:meu bem|minha linda|minha querida)\b\s*/giu,'')
  .replace(/\bvc\b/giu,'você').replace(/\bvcs\b/giu,'vocês')
  .replace(/R\$\s*(\d+)\.(\d{2})(?!\d)/g,'R$$$1,$2')
  .replace(/\s+([,.!?])/g,'$1').replace(/,([.!?])/g,'$1').replace(/([.!?])\1+/g,'$1').trim();

 return t;
}
export function finalizeCommunication(parts,profile,history=[]){
 const rules=new Set(editorialRules(profile)),out=[];let questionUsed=false;
 for(const p of parts){
  if(p.tipo!=='texto'){out.push({...p});continue;}
  if(p.purpose==='followup_question'){
   const repeated=history.some(r=>r.direction==='outbound'&&r.raw_payload?.enviado===true&&norm(r.text_content)===norm(p.texto));
   if(rules.has('no_repeated_questions')&&repeated||rules.has('one_useful_question')&&questionUsed)continue;
   questionUsed=true;
  }
  const next={...p,texto:editorialText(p.texto),communicationRevision:COMMUNICATION_REVISION};
  if(!next.texto)continue;
  const last=out.at(-1);
  // Keep photo/caption pairs, product boundaries and optional questions intact.
  const merge=rules.has('short_paragraphs')&&last?.tipo==='texto'&&last.produtoId===next.produtoId&&
   !last.allowedQuestion&&!next.allowedQuestion&&!['courtesy','product'].includes(last.purpose)&&!['courtesy','product'].includes(next.purpose)&&
   last.texto.length+next.texto.length<900;
  if(merge){if(last.texto!==next.texto)last.texto+='\n'+next.texto;}
  else out.push(next);
 }
 return out;
}

  if(sameColorSizes.length)return base+` Em ${c}, temos ${sizeLocation(sameColorSizes)}.`;
  if(cs.length&&cs.length<=6)return base+` No ${sizeWord(s)}, temos em ${joinList(cs)}.`;
  return base;
 }
 if(s)return `No momento, esse modelo está indisponível no ${sizeWord(s)}.`+(allSizes.length?` Temos ${sizeLocation(allSizes)}.`:'');
 if(c){const cs=allColors.map(lowerColor);return `No momento, esse modelo está indisponível em ${c}.`+(cs.length&&cs.length<=6?` Temos em ${joinList(cs)}.`:'');}
 return 'No momento, esse modelo está indisponível.';
}
export function sizeLocation(sizes){const ph=sizesPhrase(sizes);return /^do /.test(ph)?ph:(new Set(sizes).size===1?'no ':'nos ')+ph;}
export function colorsText(colors,size=null){
 const cs=colors.map(lowerColor);
 if(!cs.length)return `No momento, esse modelo está indisponível${size?' no '+sizeWord(size):''}.`;
 return (size?`No ${sizeWord(size)}, temos`:'Temos')+(cs.length===1?' apenas':'')+` em ${joinList(cs)}.`;
}
export function sizesText(sizes,color=null){
 if(!sizes.length)return `No momento, esse modelo está indisponível${color?' em '+lowerColor(color):''}.`;
 return (color?`Em ${lowerColor(color)}, temos`:'Temos')+' '+sizeLocation(sizes)+'.';
}

// ---- Legenda de foto (galeria) ------------------------------------------------
// De "Tamanhos disponíveis: marrom: G; marsala: GG.\nPreço: R$ 459,99." para
// "Marrom no G, marsala no GG\nR$459,99". A ordem foto → legenda é mantida.
export function humanizeCaption(text){
 const t=String(text??'').replace(new RegExp(String.fromCharCode(160),'g'),' ');
 const price=t.match(/Preço:\s*R\$\s*([\d.]+,\d{2})/);
 const sizesLine=t.match(/Tamanhos disponíveis(?: em ([^:]+))?:\s*([^\n]+?)\.?\s*(?:\n|$)/);
 if(!price||!sizesLine)return null;
 let lead;
 if(sizesLine[1]){lead=capitalize(lowerColor(sizesLine[1]))+' '+withNo(sizesLine[2]);}
 else{const groups=sizesLine[2].split(';').map(x=>x.trim()).filter(Boolean).map(g=>{const [color,sizes]=g.split(':').map(x=>x.trim());return sizes?lowerColor(color)+' '+withNo(sizes):g;});lead=capitalize(groups.join(', '));}
 return lead+'\n'+'R$'+price[1];
}
function withNo(sizes){return sizeLocation(sizes.split(/\s*(?:,|\be\b)\s*/).map(x=>x.trim()).filter(Boolean));}

// Frases do motor antigo que soam automáticas e não acrescentam fato.
const DROP_LINES=[/^encontrei .*com estoque\.?$/i,/^tenho mais modelos tamb[eé]m\.?.*$/i,/^essas fotos mostram os modelos\..*$/i,/^separei estes modelos.*$/i,/^gostou (?:desse modelo|de algum desses modelos)\??$/i,/^quer ver mais.*$/i,/^posso ajudar com outra cor ou tamanho.*$/i,/^você se interessou por (?:esse|este) modelo\??$/i];
export function dropAutomaticLines(text){
 return String(text??'').split('\n').filter(line=>!DROP_LINES.some(rx=>rx.test(line.trim()))).join('\n').trim();
}
// Afirmação negativa sobre categoria inteira ("não temos vestidos no 48") só é
// verificável para uma peça identificada; no modo geral o cadastro pode estar incompleto.
export function categoryNegativeClaim(text){return /\b(?:n[aã]o temos|n[aã]o tenho|indispon[ií]vel|esgotad[oa]s?|n[aã]o trabalhamos|n[aã]o vendemos)\b/i.test(String(text??''));}

export function referenceLead(index,total,kind){if(total===1)return '';return kind==='story'?`O ${ORD_M[index]}: `:`A ${ORD_F[index]}: `;}

// ---- Pós-processamento final das partes ---------------------------------------
export function applyVoice(parts,{greet=false,customerText='',standard=DEFAULT_STANDARD}={}){
 const out=[];
 for(const p of parts){
  if(p.tipo!=='texto'){out.push({...p});continue;}
  let t=String(p.texto??'').replace(new RegExp(String.fromCharCode(160),'g'),' ');
  const caption=/Tamanhos disponíveis/.test(t)&&/Preço:/.test(t)?humanizeCaption(t):null;
  if(caption)t=caption;
  else{
   t=dropAutomaticLines(t);
   t=t.replace(/R\$\s+(\d)/g,'R$$$1');
   t=t.replace(/^(?:Custa|O vestido custa|O conjunto custa|A peça custa|Essa peça custa)\s+(R\$[\d.]+,\d{2})\.?/gm,'$1');
  }
  if(!t.trim())continue;
  out.push({...p,texto:t,voiceRevision:VOICE_REVISION});
 }
 const first=out.find(p=>p.tipo==='texto');
 if(greet&&standard.greeting&&first&&first.purpose!=='courtesy'&&!/^(?:ol[aá]|oi|bom dia|boa tarde|boa noite)\b/i.test(first.texto))first.texto=mirroredGreeting(customerText)+'\n\n'+first.texto;
 return out;
}
